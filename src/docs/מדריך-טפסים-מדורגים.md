# מדריך טפסים מדורגים (Form Steps) במערכת

## סקירת הארכיטקטורה

מערכת הטפסים המדורגים מספקת חוויית משתמש משופרת ליצירת ועריכת נתונים מורכבים. הרכיבים מאפשרים פיצול טופס גדול למספר שלבים קטנים וברורים.

## 1. רכיבי הליבה

### 1.1 יבוא התלויות הנדרשות
```javascript
import { StepsProvider } from "src/components/steps-form/steps-provider";
import { MasterStep } from "src/components/steps-form/dynamiv-component";
import { z } from "zod";
import { Field } from "src/components/hook-form";
```

### 1.2 מבנה הטופס המדורג
```javascript
<StepsProvider
  steps={stepsArray}
  defaultValues={initialValues}
  WizardSchema={validationSchema}
  onSubmit={handleSubmit}
  watch={watchFunction}
  watchName="fieldToWatch"
/>
```

### 3.1 מבנה שדה בסיסי

## 4. וולידציה עם Zod

### 4.1 הגדרת סכמת וולידציה
```javascript
const WizardSchema = z.object({
  // שדות שלב 1
  name: z.string().min(1, "שם חובה"),
  email: z.string().email("כתובת אימייל לא תקינה"),
  
  // שדות שלב 2
  category: z.string().min(1, "קטגוריה חובה"),
  priority: z.string().optional(),
  
  // שדות שלב 3
  enabled: z.boolean(),
  confirmed: z.boolean()
});
```

### 4.2 וולידציה מותנית
```javascript
const ConditionalSchema = z.object({
  type: z.string().min(1),
  options: z.array(z.object({
    value: z.string().min(1).optional(),
    label: z.string().min(1).optional(),
  })).optional(),
  // שדות נוספים מותנים בסוג

  if (data.type === 'select' && (!data.options || data.options.length === 0)) {
    return false;
  }
  return true;
}, {
  message: "אפשרויות בחירה נדרשות עבור סוג בחירה",
```

## 7. רכיבים מותאמים אישית

### 7.1 שלב עם אפשרויות בחירה
```javascript
export function SelectOptionsStep() {
  const { control } = useFormContext();
  const { fields, append, remove } = useFieldArray({
    control,
    name: "options",
  });

  return (
    <Stack spacing={3}>
      <Alert severity="info">
        <Typography variant="body2">
          הוסף או הסר אפשרויות בחירה.
        </Typography>
      </Alert>
      
      <Scrollbar sx={{ maxHeight: { xs: 100, sm: 200, md: 260 } }}>
        <Stack spacing={2}>
          {fields.map((item, index) => (
            <Stack key={item.id} direction="row" spacing={2} alignItems="center">
              <Field.Text
                name={`options[${index}].value`}
                placeholder="ערך האפשרות"
                fullWidth
              />
              <Field.Text
                name={`options[${index}].label`}
                placeholder="תווית האפשרות"
                fullWidth
              />
              <IconButton onClick={() => remove(index)} color="error">
                <Iconify icon="mdi:delete" />
              </IconButton>
            </Stack>
          ))}
        </Stack>
      </Scrollbar>
      
      <Button
        type="button"
        variant="outlined"
        startIcon={<Iconify icon="mdi:plus" />}
        onClick={() => append({ value: '', label: '' })}
      >
        הוסף אפשרות
      </Button>
    </Stack>
  );
}
```

## 8. דיאלוג מלא

### 8.1 דיאלוג עם טופס מדורג
```javascript
export function SteppedFormDialog({ open, onClose, initialData, onComplete }) {
  const handleWizardComplete = (data) => {
    if (onComplete) {
      onComplete(data);
    }
    onClose();
  };

  return (
    <Dialog open={open} onClose={onClose} maxWidth="md" fullWidth>
      <StepsProvider
        steps={stepsArray}
        defaultValues={initialData}
        WizardSchema={validationSchema}
        onSubmit={handleWizardComplete}
        watch={watchFunction}
        watchName="type"
      />
    </Dialog>
  );
}
```

## 9. דוגמת יישום מלאה

### 9.1 טופס עריכת עמודה (מהמערכת)
```javascript
export function ColumnDefinitionStep({ column, onComplete }) {
  const initialValues = {
    name: column.name,
    label: column.label,
    type: column.type,
    options: column.options || [],
    hidden: column.hidden === '',
    required: column.required === '',
    group: column.filters,
    priority: column.group_name,
  };

  const fields = [
    // שלב 1: פרטים בסיסיים
    {
      step: 1,
      name: "name",
      label: "שם עמודה",
      variant: "filled",
      InputLabelProps: { shrink: true },
      type: "text",
      component: Field.Text
    },
    {
      step: 1,
      name: "label",
      label: "תווית עמודה",
      variant: "filled",
      InputLabelProps: { shrink: true },
      type: "text",
      component: Field.Text
    },
    // שלב 2: סוג נתונים
    {
      step: 2,
      name: "type",
      label: "סוג נתונים",
      variant: "filled",
      InputLabelProps: { shrink: true },
      children: Object.values(columnsTypes).map(type => (
        <MenuItem key={type.type} value={type.type}>
          <Iconify icon={type.icon} width={20} />
          <Typography variant="body2">{type.label}</Typography>
        </MenuItem>
      )),
      component: Field.Select
    },
    // שלב 3: אפשרויות (מותנה)
    {
      step: 3,
      name: "options",
      label: "אפשרויות",
      component: SelectOptionsStep
    },
    // שלב 4: הגדרות תצוגה
    {
      step: 4,
      name: 'hidden',
      label: 'העמודה מוסתרת',
      icon: 'mdi:eye-off-outline',
      component: Field.Switch
    },
    {
      step: 4,
      name: 'required',
      label: 'העמודה חובה',
      icon: 'mdi:check-circle-outline',
      component: Field.Switch
    }
  ];

  const createSteps = (includeOptions = false) => [
    {
      label: 'פרטים בסיסיים',
      component: <MasterStep fields={fields} number={1} />,
      icon: "mdi:account-outline",
      name: 'basicDetails'
    },
    {
      label: 'סוג נתונים',
      component: <MasterStep fields={fields} number={2} />,
      icon: "mdi:database-outline",
      name: 'dataType'
    },
    ...(includeOptions ? [{
      label: 'אפשרויות בחירה',
      component: <MasterStep fields={fields} number={3} />,
      icon: "mdi:format-list-bulleted-type",
      name: 'selectOptions'
    }] : []),
    {
      label: 'הגדרות תצוגה',
      component: <MasterStep fields={fields} number={4} />,
      icon: "mdi:palette-outline",
      name: 'displaySettings'
    },
    {
      name: 'complete',
      component: <></>
    }
  ];

  const [steps, setSteps] = useState(createSteps(false));

  const updateSteps = useCallback((data) => {
    setSteps(createSteps(data.type === 'select'));
  }, []);

  return (
    <StepsProvider
      steps={steps}
      defaultValues={initialValues}
      WizardSchema={WizardSchema}
      onSubmit={onComplete}
      watch={updateSteps}
      watchName='type'
    />
  );
}
```

## 11. טיפים חשובים

### 11.1 ביצועים
- השתמש ב-`useCallback` לפונקציות שמועברות כ-props
- הימנע מ-re-render מיותר של שלבים

### 11.2 נגישות
- השתמש בתוויות עבריות ברורות
- הוסף הסברים לשדות מורכבים
- וודא תמיכה בניווט מקלדת

### 11.3 חוויית משתמש
- הוסף הודעות שגיאה ברורות
- שמור מצב בין שלבים
- אפשר חזרה לשלבים קודמים

---

מערכת הטפסים המדורגים מספקת חוויית משתמש מעולה ומאפשרת יצירת טפסים מורכבים בצורה מאורגנת ונוחה למשתמש.
