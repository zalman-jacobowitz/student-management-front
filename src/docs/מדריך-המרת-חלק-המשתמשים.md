# מדריך המרת חלק המשתמשים למבנה החדש

## סקירה כללית

מדריך זה מספק תכנית מפורטת להמרת חלק המשתמשים (`src/sections/users/`) מהארכיטקטורה הנוכחית לארכיטקטורה החדשה המבוססת על `FullTableWrapper` ו-`StepsProvider`.

## 1. ניתוח המבנה הנוכחי

### 1.1 קבצים קיימים
```
src/sections/users/
├── componenets/
│   └── button-icon-form.jsx          # רכיב כפתורי טוגל
```

- **עקביות**: לא עקבי עם חלקי המערכת האחרים
- **תחזוקה**: קשה לתחזוקה והרחבה

```javascript
// view/users-list-view.tsx
import { useSuspenseQuery, useQueryClient, useMutation } from "@tanstack/react-query";
import { Suspense, useCallback, useMemo } from "react";

import { paths } from "src/routes/paths";
import { apiUsers, usersUpdate } from "src/actions/users";
import { LoadingScreen } from "src/components/loading-screen";
import { FullTableWrapper } from "src/components/full-table/view";
import { TableColumnsConfig } from "src/components/full-table/types";
import { toast } from "sonner";
import { useRouter } from "src/routes/hooks";
import { uuidv4 } from "src/utils/uuidv4";
import { useUserDetails } from "src/hooks/use-user-details";

import { USERS_COLUMNS } from "../users-columns";
import { UsersEditDialog } from "../components/users-edit-dialog";

const LINKS = [
  { name: 'מסך-ראשי', href: paths.dashboard.root },
  { name: 'משתמשים', href: paths.dashboard.users },
  { name: 'רשימה' },
];

function UsersMainView() {
  const usersData = useSuspenseQuery(apiUsers());
  const router = useRouter();
  const queryClient = useQueryClient();
  const mutate = useMutation(usersUpdate({ queryClient }));
  
  const submitDelete = useCallback(async (data: any[]) => {
    try {
      const promise = mutate.mutateAsync({
        data: data.map(item => ({ user_id: item })), 
        mode: 'delete'
      });

      toast.promise(promise, {
        loading: 'מחיקה...',
        success: 'המחיקה הצליחה!',
        error: 'המחיקה נכשלה!',
      });

      await promise;
    } catch (error) {
      console.error('Submission error:', error);
      toast.error('שגיאה בשליחת הטופס');
    }
  }, [mutate]);

  const listActionsMap = [
    {
      icon: 'solar:settings-bold',
      label: 'הגדרות',
      onClick: () => {
        router.push(paths.dashboard.infoColumns);
      }
    },
    {
      icon: 'solar:export-bold',
      label: 'ייצוא',
      onClick: () => {
        // exportToExcel([], 'users_export.xlsx');
      }
    }
  ];

  const userDetails = useUserDetails();

  const configUsers: TableColumnsConfig = useMemo(() => ({
    heading: {
      links: LINKS,
      title: 'רשימת משתמשים',
      importButton: true,
    },
    toolbar: {
      listActionsMap: listActionsMap,
      removeAction: true,
      onDelete: (selected) => {
        submitDelete(selected);
      }
    },
    data: {
      tableData: usersData.data,
      tableColumns: USERS_COLUMNS,
    },
    rowDetails: {
      specialRow: ['checkbox', 'avatar', 'edit'],
      id: 'user_id',
      EditComponent: (props) => {
        const { open, onClose, column } = props;

        return (
          <UsersEditDialog
            user={column}
            open={open}
            onClose={onClose}
            existingUsers={usersData.data || []}
            isEditing
          />
        );
      }
    },
    tableDetails: {
      styleTable: 'default',
      pagination: true,
      addButton: true,
      defaultValues: {
        user_id: uuidv4(),
        created_by: userDetails?.userDetails?.email || '',
        client: 'kg_gdola',
      }
    }
  }), [usersData.data, submitDelete, userDetails, listActionsMap]);

  return <FullTableWrapper config={configUsers} />;
}

export function UsersViewWrapper() {
  return (
    <Suspense fallback={<LoadingScreen />}>
      <UsersMainView />
    </Suspense>
  );
}
```

```javascript
// components/users-edit-dialog.jsx
import { useCallback } from "react";
import { Dialog } from "@mui/material";
import { toast } from "sonner";
import { useMutation, useQueryClient } from "@tanstack/react-query";

import { StepsProvider } from "src/components/steps-form/steps-provider";
import { usersUpdate } from "src/actions/users";

import { createUserSteps, USER_FORM_FIELDS } from "../users-steps-config";
import { UserFormSchema } from "./users-validation";

export function UsersEditDialog({ 
  open, 
  onClose, 
  user, 
  existingUsers = [], 
  isEditing = false 
}) {
  const queryClient = useQueryClient();
  const mutate = useMutation(usersUpdate({ queryClient }));

  const initialValues = {
    firstName: user?.firstName || '',
    lastName: user?.lastName || '',
    email: user?.email || '',
    country: user?.country || '',
    screens: user?.screens || {},
    permissions: user?.permissions || [],
    verified: user?.verified || false
  };

  const handleSubmit = useCallback(async (data) => {
    try {
      const userDetails = {
        user_id: user?.user_id || data.user_id,
        firstName: data.firstName,
        lastName: data.lastName,
        email: data.email,
        country: data.country,
        screens: data.screens,
        permissions: data.permissions,
        verified: data.verified,
        client: user?.client || 'kg_gdola'
      };

      const promise = mutate.mutateAsync({
        data: [userDetails],
        mode: isEditing ? 'update' : 'create'
      });

      toast.promise(promise, {
        loading: isEditing ? 'מעדכן משתמש...' : 'יוצר משתמש...',
        success: isEditing ? 'המשתמש עודכן בהצלחה' : 'המשתמש נוצר בהצלחה',
        error: isEditing ? 'שגיאה בעדכון המשתמש' : 'שגיאה ביצירת המשתמש'
      });

      await promise;
      onClose();
      return true;
    } catch (error) {
      console.error('Error saving user:', error);
      return false;
    }
  }, [user, mutate, isEditing, onClose]);

  const steps = createUserSteps();

  return (
    <Dialog 
      open={open} 
      onClose={onClose} 
      maxWidth="md" 
      fullWidth
    >
      <StepsProvider
        steps={steps}
        defaultValues={initialValues}
        WizardSchema={UserFormSchema}
        onSubmit={handleSubmit}
      />
    </Dialog>
  );
}
```

#### 3.7 שלב הרשאות מעודכן
```javascript
// components/users-permissions-step.jsx
import { useState } from 'react';
import { useFormContext } from 'react-hook-form';
import { 
  Typography, 
  Button, 
  Card, 
  Box, 
  IconButton, 
  alpha 
} from '@mui/material';

import { Iconify } from 'src/components/iconify';
import { AddPermissionDialog } from './add-permission-dialog';

export function UsersPermissionsStep() {
  const { watch, setValue } = useFormContext();
  const filters = watch('permissions') || [];
  const [open, setOpen] = useState(false);

  const addFilter = (table, column, value, operator, side) => {
    const next = [...filters];
    const existing = next.find((f) => f.table === table && f.columnName === column);
    
    if (existing) {
      existing.values = Array.from(new Set([...(existing.values ?? []), value]));
      existing.operator = operator;
      existing.side = side;
    } else {
      next.push({
        table,
        columnName: column,
        values: [value],
        operator,
        side,
      });
    }
    setValue('permissions', next);
  };

  const removeFilter = (table, column) => {
    setValue(
      'permissions',
      filters.filter((f) => !(f.table === table && f.columnName === column))
    );
  };

  return (
    <>
      <Typography variant="subtitle1" mb={2}>
        הגדרת הרשאות גישה
      </Typography>
      <Typography variant="body2" color="text.secondary" mb={3}>
        קבע אילו רשומות יוכל המשתמש לראות ולערוך
      </Typography>

      <Card sx={{ p: 3, mb: 3, bgcolor: 'background.neutral' }}>
        {filters.length ? (
          <>
            <Typography variant="subtitle2" mb={2}>
              הגבלות פעילות ({filters.length})
            </Typography>
            <Box sx={{ display: 'flex', flexDirection: 'column', gap: 1.5 }}>
              {filters.map((filter, index) => (
                <Box
                  key={index}
                  sx={{
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'space-between',
                    p: 2,
                    borderRadius: 1.5,
                    bgcolor: 'background.paper',
                    border: (t) => `1px solid ${t.palette.divider}`,
                  }}
                >
                  <Box sx={{ flex: 1 }}>
                    <Typography variant="body2" fontWeight="medium">
                      {filter.table} - {filter.columnName}: {filter.values.join(', ')}
                    </Typography>
                    <Typography variant="caption" color="text.secondary">
                      {filter.operator.toUpperCase()} | {filter.side === 'server' ? 'שרת' : 'לקוח'}
                    </Typography>
                  </Box>
                  <IconButton
                    size="small"
                    color="error"
                    onClick={() => removeFilter(filter.table, filter.columnName)}
                  >
                    <Iconify icon="solar:trash-bin-trash-bold" />
                  </IconButton>
                </Box>
              ))}
            </Box>
          </>
        ) : (
          <Box textAlign="center">
            <Iconify
              icon="solar:shield-user-bold"
              sx={{ width: 48, height: 48, color: 'text.disabled', mb: 2 }}
            />
            <Typography variant="h6" color="text.secondary">
              לא הוגדרו הגבלות
            </Typography>
            <Typography variant="body2" color="text.disabled">
              המשתמש יוכל לראות את כל הרשומות
            </Typography>
          </Box>
        )}
      </Card>

      <Button
        fullWidth
        size="large"
        variant="contained"
        startIcon={<Iconify icon="solar:add-circle-bold" />}
        onClick={() => setOpen(true)}
      >
        הוסף הגבלה
      </Button>

      <AddPermissionDialog
        open={open}
        onClose={() => setOpen(false)}
        onAdd={addFilter}
        existingFilters={filters}
      />
    </>
  );
}
```
